<div id="consulta-ruc" style="max-width:600px; margin:40px auto; font-family:Arial, sans-serif;">
  <h3 style="color:#1a3c34;">Consulta de RUC</h3>
  <p>Ingresá el RUC (ej: 80012345-6) y consultá los datos del contribuyente.</p>
  <div style="display:flex; gap:10px; margin-bottom:15px;">
    <input type="text" id="ruc-input" placeholder="Ej: 80012345-6"
           style="flex:1; padding:12px; border:1px solid #ccc; border-radius:6px; font-size:16px;">
    <button onclick="consultarRUC()"
            style="padding:12px 20px; background:#1a3c34; color:white; border:none; border-radius:6px; font-size:16px; cursor:pointer;">
      Consultar
    </button>
  </div>
  <div id="ruc-resultado" style="background:#f5f5f5; padding:15px; border-radius:6px; display:none;"></div>
</div>

<script>
async function consultarRUC() {
  const ruc = document.getElementById('ruc-input').value.trim();
  const resultado = document.getElementById('ruc-resultado');

  if (!ruc) {
    alert('Por favor ingresá un RUC');
    return;
  }

  resultado.style.display = 'block';
  resultado.innerHTML = '<p>Cargando...</p>';

  try {
    const resp = await fetch(`TU_URL_BACKEND/api/ruc?ruc=${encodeURIComponent(ruc)}`);
    const data = await resp.json();

    if (data.error) {
      resultado.innerHTML = `<p style="color:red;">Error: ${data.error}</p>`;
    } else {
      resultado.innerHTML = `
        <p><strong>RUC:</strong> ${data.ruc || '-'}</p>
        <p><strong>Razón Social:</strong> ${data.razonSocial || '-'}</p>
        <p><strong>Nombre Fantasía:</strong> ${data.nombreFantasia || '-'}</p>
        <p><strong>Estado:</strong> ${data.estado || '-'}</p>
        <p><strong>Categoría:</strong> ${data.categoria || '-'}</p>
      `;
    }
  } catch (err) {
    resultado.innerHTML = '<p style="color:red;">No se pudo conectar al servidor. Verificá que el backend esté activo.</p>';
  }
}
</script>
